const texturePaths:Record<string,string>={'fire-tongues-v1':'vfx/production/fire-tongues-v1','ribbon-white-v1':'vfx/production/ribbon-white-v1','ribbon-soft-v1':'vfx/production/ribbon-soft-v1'};
export const vfxColor=(hex:string,alpha=255)=>{const c=new Color().fromHEX(hex);c.a=alpha;return c;};
export function constant(curve:CurveRange,value:number){curve.mode=CurveRange.Mode.Constant;curve.constant=value;}
function range(curve:CurveRange,min:number,max:number){curve.mode=CurveRange.Mode.TwoConstants;curve.constantMin=min;curve.constantMax=max;}
function curve(curve:CurveRange,keys:[number,number][]){curve.mode=CurveRange.Mode.Curve;curve.spline.assignSorted(keys.map(([time,value])=>[time,{value}] as [number,{value:number}]));}
function gradient(out:GradientRange,colors:[number,string][],alpha:[number,number][]){out.mode=GradientRange.Mode.Gradient;out.gradient.setKeys(colors.map(([time,color])=>({time,color:vfxColor(color)})),alpha.map(([time,alpha])=>({time,alpha:Math.round(alpha*255)})));}
/** Native module classes are registered by ParticleSystem. Configure them before
 * activating the node so Cocos onLoad binds and initializes them, as with a prefab. */
function nativeModule<T>(name:string):T{const ctor=js.getClassByName(name) as new()=>T;if(!ctor)throw new Error('Cocos particle module unavailable: '+name);return new ctor();}

/** CC0 masks and separately documented generated RGBA with built-in shaders. */
export class VfxAssets {
 private static textures=new Map<string,Texture2D>();
 private static materials=new Map<string,Material>();
 private static pending:Promise<void>|null=null;
 static preload():Promise<void>{
  if(this.pending)return this.pending;
  this.pending=Promise.all(textures.map(name=>new Promise<void>((resolve,reject)=>resources.load((texturePaths[name]??'vfx/kenney/'+name)+'/texture',Texture2D,(error,texture)=>{if(error){reject(new Error('VFX '+name+': '+error.message));return;}texture.setWrapMode(Texture2D.WrapMode.CLAMP_TO_EDGE,Texture2D.WrapMode.CLAMP_TO_EDGE);if(name.startsWith('ribbon-')){texture.setFilters(Texture2D.Filter.LINEAR,Texture2D.Filter.LINEAR);texture.setMipFilter(Texture2D.Filter.NONE);}this.textures.set(name,texture);resolve();})))).then(()=>undefined).catch(error=>{this.dispose();throw error;});return this.pending;
 }
 static material(texture:string,shader:'particle'|'line'='particle',alphaBlend=false,tintGain=1,alphaGain=1):Material {
  // builtin-particle-trail defaults to 119/255 grey. A colour-only Line must
  // sample explicit white or the alleged bright core loses 53% of its RGB.
  if(shader==='line'&&!texture)texture='ribbon-white-v1';
  const key=texture+shader+alphaBlend+':'+tintGain+':'+alphaGain;const cached=this.materials.get(key);if(cached)return cached;
  const material=new Material();material.name=(texture.startsWith('ribbon-')?'ProjectProcedural:':texturePaths[texture]?'OpenAIImageGen:':'KenneyCC0:')+key;
  // Creator 3.8 registers these built-ins under their particles/ asset folder.
  // A short name silently creates a material with no passes, crashing the CPU renderer.
  // The built-in trail alpha technique expects premultiplied RGB, whereas
  // native Line supplies straight vertex colors. Keep its shader and explicitly
  // select straight-alpha blending for solid electric ribbons. Glow uses the
  // built-in additive technique and a separate soft alpha cross section.
  material.initialize({effectName:shader==='line'?'particles/builtin-particle-trail':'particles/builtin-particle',technique:alphaBlend?1:0,
   ...(shader==='line'&&alphaBlend?{states:{blendState:{targets:[{blend:true,blendSrc:gfx.BlendFactor.SRC_ALPHA,blendDst:gfx.BlendFactor.ONE_MINUS_SRC_ALPHA,blendSrcAlpha:gfx.BlendFactor.ONE,blendDstAlpha:gfx.BlendFactor.ONE_MINUS_SRC_ALPHA}]}}}:{})});
  material.setProperty('tintColor',new Vec4(.5*tintGain,.5*tintGain,.5*tintGain,.5*alphaGain));
  if(texture){const image=this.textures.get(texture);if(!image)throw new Error('VfxAssets.preload must complete before '+texture);material.setProperty('mainTexture',image);}
  this.materials.set(key,material);return material;
 }
 static lineMaterial(style:VfxLineStyle,electric=false):Material {
  if(style==='glow')return this.material('ribbon-soft-v1','line',false,1.65);
  // Blue lightning uses a coloured hot filament rather than exposure-clipped
  // white. Other weapon/arrow cores retain their existing metallic highlight.
  if(style==='core')return this.material('ribbon-white-v1','line',true,electric?1.05:1.25);
  return this.material('ribbon-white-v1','line',electric,electric?1.1:1);
 }
 static lineTexture(style:VfxLineStyle):Texture2D {
  const name=style==='glow'?'ribbon-soft-v1':'ribbon-white-v1',texture=this.textures.get(name);
  if(!texture)throw new Error('VfxAssets.preload must complete before '+name);
  return texture;
 }
 static solid(color:string):Material{
  const key='solid'+color;const cached=this.materials.get(key);if(cached)return cached;
  const material=new Material();material.initialize({effectName:'builtin-standard'});material.setProperty('mainColor',vfxColor(color));material.setProperty('roughness',.52);material.setProperty('metallic',color==='#b8bcb8'?.65:0);this.materials.set(key,material);return material;
 }
 static reset(ps:ParticleSystem,kind:VfxKind){const p=profiles[kind];ps.duration=10;ps.loop=true;ps.bursts=[];constant(ps.rateOverTime,p.rate);ps.startColor.color=vfxColor(p.color,Math.round(p.alpha*255));if(p.spin)range(ps.startRotationZ,-Math.PI,Math.PI);else constant(ps.startRotationZ,0);}
 static emitter(kind:VfxKind,parent:Node):{node:Node;system:ParticleSystem;life:number}{
  const p=profiles[kind],node=new Node('VFX:'+kind);node.active=false;parent.addChild(node);
  const ps=node.addComponent(ParticleSystem);ps.playOnAwake=false;ps.loop=true;ps.duration=10;ps.capacity=p.capacity;ps.simulationSpace=0;ps.scaleSpace=1;
  range(ps.startLifetime,...p.life);range(ps.startSizeX,...p.size);range(ps.startSpeed,...p.speed);constant(ps.rateOverTime,p.rate);constant(ps.rateOverDistance,0);constant(ps.gravityModifier,p.gravity??0);constant(ps.startDelay,0);
  if(p.spin)range(ps.startRotationZ,-Math.PI,Math.PI);else constant(ps.startRotationZ,0);
  ps.startColor.color=vfxColor(p.color,Math.round(p.alpha*255));
  const shape=nativeModule<NonNullable<ParticleSystem['shapeModule']>>('cc.ShapeModule');shape.shapeType=p.shape;shape.radius=p.radius;shape.angle=18;shape.length=.15;if(p.shape===2)shape.emitFrom=0;shape.rotation=new Vec3(90,0,0);shape.enable=true;ps.shapeModule=shape;
  const color=nativeModule<NonNullable<ParticleSystem['colorOverLifetimeModule']>>('cc.ColorOvertimeModule');color.enable=true;gradient(color.color,[[0,'#ffffff'],[p.fast ? .3 : .5,'#ffffff'],[1,p.end]],p.fast?[[0,0],[.08,1],[.38,.9],[.6,.32],[1,0]]:[[0,0],[.08,1],[.5,.85],[1,0]]);ps.colorOverLifetimeModule=color;
  const size=nativeModule<NonNullable<ParticleSystem['sizeOvertimeModule']>>('cc.SizeOvertimeModule');size.enable=true;curve(size.size,p.grow?[[0,.55],[.45,1],[1,1.4]]:[[0,.8],[.15,1],[1,.12]]);ps.sizeOvertimeModule=size;
  ps.renderer.renderMode=p.render??0;ps.renderer.velocityScale=.08;ps.renderer.lengthScale=2;
  // OnInit establishes the renderer's particle-system owner after activation.
  node.active=true;ps.setSharedMaterial(this.material(p.texture,'particle',p.blend,p.tintGain,p.alphaGain),0);ps.stop();this.reset(ps,kind);
  return {node,system:ps,life:p.life[1]};
 }
 static dispose(){for(const material of this.materials.values())material.destroy();this.materials.clear();for(const texture of this.textures.values())assetManager.releaseAsset(texture);this.textures.clear();this.pending=null;}
}
